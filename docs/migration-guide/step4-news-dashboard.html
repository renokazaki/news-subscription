Step 4: ニュースデータ & ダッシュボード

所要時間: 1〜2日 | フィルタリングAPI + レイアウト構成

このStepのゴール

ニュース一覧がフィルタリング（日付・Interest）付きで表示され、詳細画面へ遷移できる。DashboardLayoutにSidebar + Header + コンテンツエリアが構成される。

4-1. News モデル Rails

rails generate model News user:references title:string text:text url:string tag:string published_at:datetime
rails db:migrate
backend/app/models/news.rb
class News < ApplicationRecord
  belongs_to :user

  validates :title, :url, presence: true

  scope :by_date, ->(date) { where('DATE(published_at) = ?', date) if date.present? }
  scope :by_tag, ->(tag) { where(tag: tag) if tag.present? }
  scope :recent, -> { order(published_at: :desc) }
end
scope はActiveRecordのクエリをメソッドチェーンで組み合わせ可能にする仕組み。条件付きscope（if date.present?）は、パラメータがない時はそのscopeを無視してくれるので、コントローラー側で条件分岐を書く必要がない。

News.recent.by_date('2024-01-01').by_tag('React') のようにチェーンでき、SQL発行は1回だけ。各scopeが独立しているのでテストも容易。
テーブル名が news（不可算名詞）のため、Railsの自動命名規則と微妙にズレる場合がある。rails generate model News で生成した場合は問題ないが、手動で作る場合は self.table_name = 'news' をモデルに明記する。

4-2. News コントローラー Rails

backend/app/controllers/api/v1/news_controller.rb
module Api
  module V1
    class NewsController < ApplicationController
      before_action :authenticate_user!

      def index
        news = current_user.news
                  .recent
                  .by_date(params[:date])
                  .by_tag(params[:tag])

        render json: news.map { |n| news_summary(n) }
      end

      def show
        news_item = current_user.news.find(params[:id])
        render json: news_detail(news_item)
      end

      private

      def news_summary(news)
        {
          id: news.id,
          title: news.title,
          tag: news.tag,
          published_at: news.published_at,
          url: news.url
        }
      end

      def news_detail(news)
        {
          id: news.id,
          title: news.title,
          text: news.text,
          tag: news.tag,
          url: news.url,
          published_at: news.published_at
        }
      end
    end
  end
end
indexとshowでレスポンス形式を分ける理由: 一覧表示では text（要約本文）は不要。大量のニュースを一覧取得する際にテキスト全文を含めるとレスポンスサイズが肥大化し、パフォーマンスが悪化する。詳細画面で初めて全文を返す設計。
フィルタリングをサーバーサイドにする理由: 旧実装ではクライアントで全件取得してからJSでフィルタしていた。データ量が少ないうちは問題ないが、ニュースが数百件になるとフロントのメモリと初回ロード時間が問題になる。DBのWHERE句は数万件でも高速に動作する。

4-3. RSpec リクエストスペック Rails

backend/spec/requests/api/v1/news_spec.rb
require 'rails_helper'

RSpec.describe 'Api::V1::News', type: :request do
  let(:user) { create(:user) }
  let(:token) { Warden::JWTAuth::UserEncoder.new.call(user, :user, nil).first }
  let(:auth_headers) { { 'Authorization' => "Bearer #{token}" } }

  describe 'GET /api/v1/news' do
    before do
      create(:news, user: user, title: 'React 19 Released', tag: 'React',
             published_at: Date.new(2024, 6, 15))
      create(:news, user: user, title: 'Rails 8 Preview', tag: 'Rails',
             published_at: Date.new(2024, 6, 14))
    end

    it 'returns all news for the current user' do
      get '/api/v1/news', headers: auth_headers, as: :json

      expect(response).to have_http_status(:ok)
      json = JSON.parse(response.body)
      expect(json.length).to eq(2)
    end

    it 'filters by tag' do
      get '/api/v1/news', params: { tag: 'React' },
          headers: auth_headers, as: :json

      json = JSON.parse(response.body)
      expect(json.length).to eq(1)
      expect(json.first['tag']).to eq('React')
    end

    it 'filters by date' do
      get '/api/v1/news', params: { date: '2024-06-15' },
          headers: auth_headers, as: :json

      json = JSON.parse(response.body)
      expect(json.length).to eq(1)
    end
  end

  describe 'GET /api/v1/news/:id' do
    let!(:news_item) { create(:news, user: user, title: 'Detail Test', text: 'Full text here') }

    it 'returns news detail including text' do
      get "/api/v1/news/#{news_item.id}",
          headers: auth_headers, as: :json

      expect(response).to have_http_status(:ok)
      json = JSON.parse(response.body)
      expect(json['title']).to eq('Detail Test')
      expect(json['text']).to eq('Full text here')
    end
  end
end

4-6. フロントエンド: OpenAPIフック React

Step 4-3のRSpecテストからOpenAPI仕様書が生成済み。$api を使って型安全にNews APIを呼び出す:

ニュース一覧取得（フィルタ付き）
import { $api } from '../lib/api/client';

// 一覧取得 — クエリパラメータも型安全
const { data: news, isLoading } = $api.useQuery(
  'get', '/api/v1/news',
  {
    params: {
      query: {
        date: selectedDate,    // string | undefined
        tag: selectedInterest, // string | undefined
      },
    },
  }
);

// 詳細取得
const { data: detail } = $api.useQuery(
  'get', '/api/v1/news/{id}',
  {
    params: { path: { id: newsId } },
  }
);
手動フックとの違い: 従来の useNews フックでは URLSearchParams の組み立てやqueryKeyの設計を手動で行っていた。$api.useQuery ではパスとパラメータを指定するだけで、型チェック・queryKey生成・URL組み立てが全て自動化される。

query パラメータに undefined の値がある場合は自動的に除外されるため、フィルタ条件の有無による分岐も不要。

フィルタ状態の管理

// Jotai atom でUIローカル状態を管理
// selectedDateAtom: カレンダーで選択した日付
// selectedInterestAtom: サイドバーで選択したInterest

// DashboardPage で $api.useQuery に渡す:
const [selectedDate] = useAtom(selectedDateAtom);
const [selectedInterest] = useAtom(selectedInterestAtom);

const { data: news, isLoading } = $api.useQuery(
  'get', '/api/v1/news',
  {
    params: {
      query: {
        date: selectedDate ? format(selectedDate, 'yyyy-MM-dd') : undefined,
        tag: selectedInterest,
      },
    },
  }
);
フィルタ状態（UI状態）は Jotai、サーバーデータ（ニュース一覧）は TanStack Query（$api 経由）で管理する。この分離が重要: Jotaiのatomが変わる → $api.useQuery のparamsが変わる → queryKeyが変わる → 自動再フェッチ。明示的に「再取得」ロジックを書く必要がない。

4-8. 動作確認チェックリスト

  1. ダッシュボードにニュース一覧が表示される
  2. カレンダーで日付選択 → その日のニュースだけ表示される
  3. サイドバーでInterest選択 → そのタグのニュースだけ表示される
  4. ニュースカードクリック → /news/:id に遷移し詳細が表示される
  5. フィルタ解除 → 全件に戻る
  6. 未認証でダッシュボードURL直アクセス → ログインページにリダイレクト
